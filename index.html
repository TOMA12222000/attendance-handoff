OneDriveのファイル数を確認

社用PCでWindows PowerShellを開き、下のコードをコピーして実行します。業務用OneDriveに同期されているファイルの件数を画面に表示します。ファイルの移動・削除や集計結果の送信は行いません。

フォルダ別の件数はサブフォルダを含みます。直下にあるTeams・Copilotの管理用と見られるフォルダは、フォルダ名で判定して Skipped と表示します。Could not count は集計エラーを示します。

整理は、集計 → ファイル名などから既存フォルダへの分類ルール作成 → 移動先候補を一覧で確認 → 承認済みルールで移動 → 新規ファイルに適用の順で進めます。Teams・Copilotなどが管理するフォルダは整理対象から外します。

OneDriveの分類候補を確認して選んだファイルを移動

  1. コードをコピーし、社用PCのWindows PowerShellに貼り付けて実行します。
  2. 案件の分類も行う場合は、略称=既存の案件フォルダ名 または既存のフォルダ名をカンマ区切りで入力します。フォルダ名が不明なら略称だけ入力し、表示された一覧から移動先の番号を入力します。案件を指定しない場合はEnterを押します。
  3. 候補一覧を確認し、移動するファイルの番号だけを 1,3,4 のように入力します。確認だけならEnterで終了します。

研修資料は「研修」、領収書は「経費」、健康保険の資格確認書は「プロフィール」が候補です。「経費」がなければ、そのフォルダへの移動を選んだときに作成します。.url などのショートカットは保留します。

対象はOneDrive直下と「ドキュメント」配下です。Teams・Copilotの名前を含むフォルダは除外します。候補が複数ある場合や同名のファイルがある場合は保留し、上書きはしません。処理は社用PC内で行い、移動したファイルにはOneDriveの同期が適用されます。

& {
    $root = $env:OneDriveCommercial
    if ([string]::IsNullOrWhiteSpace($root) -or
        -not (Test-Path -LiteralPath $root -PathType Container)) {
        throw '業務用OneDriveの同期フォルダが見つかりません。'
    }
    $root = (Get-Item -LiteralPath $root -ErrorAction Stop).FullName
    $managedFolder = '(?i)Teams|Copilot'
    $rootFolders = @(Get-ChildItem -LiteralPath $root -Directory -ErrorAction Stop |
        Where-Object { $_.Name -notmatch $managedFolder -and -not $_.LinkType })
    $documents = $rootFolders | Where-Object Name -eq 'ドキュメント' | Select-Object -First 1

    $specialFolders = @('研修', '経費', 'プロフィール')
    $resolveAliases = {
        param($entries, $availableFolders)

        $builtIn = @('研修', '経費', 'プロフィール')
        $destinations = @($availableFolders | Where-Object {
            $_.Name -notin (@('ドキュメント', 'Documents') + $builtIn)
        } | Sort-Object Name)
        $confirmed = foreach ($entry in ($entries -split '[,、，]' | Where-Object { $_.Trim() })) {
            $parts = $entry.Trim() -split '[=＝]', 2
            $keyword = $parts[0].Trim()
            $requestedName = $parts[-1].Trim()
            if ($parts.Count -eq 1 -and $keyword -in $builtIn) { continue }
            if (-not $keyword) {
                Write-Warning "キーワードが空のルールはスキップしました：$entry"
                continue
            }

            $folder = $destinations | Where-Object Name -eq $requestedName | Select-Object -First 1
            if ($null -eq $folder) {
                if ($destinations.Count -eq 0) {
                    Write-Warning "選べる案件フォルダがありません：$keyword"
                    continue
                }
                Write-Host ("「{0}」の移動先を、実際のOneDriveフォルダから選んでください。" -f $keyword)
                for ($i = 0; $i -lt $destinations.Count; $i++) {
                    Write-Host ("{0}. {1}" -f ($i + 1), $destinations[$i].Name)
                }
                $choiceText = Read-Host '移動先の番号（Enterはこの案件ルールをスキップ）'
                if ([string]::IsNullOrWhiteSpace($choiceText)) { continue }
                $choice = 0
                if (-not [int]::TryParse($choiceText, [ref]$choice) -or
                    $choice -lt 1 -or $choice -gt $destinations.Count) {
                    Write-Warning "無効な番号のため、案件ルールをスキップしました：$keyword"
                    continue
                }
                $folder = $destinations[$choice - 1]
            }

            if ($folder.Name -match '[,、，]') {
                Write-Warning "カンマを含むフォルダ名は案件ルールに登録できません：$($folder.Name)"
                continue
            }
            Write-Host ("案件ルール：{0} → {1}" -f $keyword, $folder.Name)
            "{0}={1}" -f $keyword, $folder.Name
        }
        $confirmed -join ','
    }
    $entryText = Read-Host '案件名や略称（例：略称=フォルダ名,フォルダ名）。省略はEnter'
    $entryText = & $resolveAliases $entryText $rootFolders
    $rules = @(
        [pscustomobject]@{ Folder = '研修'; Pattern = '研修|トレーニング|(?<![a-z])training(?![a-z])' }
        [pscustomobject]@{ Folder = '経費'; Pattern = '領収書|(?<![a-z])receipt(?![a-z])' }
        [pscustomobject]@{ Folder = 'プロフィール'; Pattern = '健康保険[ _　-]*資格確認書' }
        foreach ($entry in ($entryText -split '[,、，]' | Where-Object { $_.Trim() })) {
            $parts = $entry.Trim() -split '=', 2
            $keyword = $parts[0].Trim()
            $name = $parts[-1].Trim()
            $folder = $rootFolders | Where-Object {
                $_.Name -eq $name -and $_.Name -notin (@('ドキュメント', 'Documents') + $specialFolders)
            } | Select-Object -First 1
            if (-not $keyword -or $null -eq $folder) {
                Write-Warning "案件ルールを登録できません：$entry"
                continue
            }
            $keywords = @($keyword, $folder.Name) | Select-Object -Unique
            foreach ($term in $keywords) {
                $pattern = [regex]::Escape($term)
                if ($term -match '^[a-z0-9]+$') {
                    $pattern = "(?<![a-z0-9])$pattern(?![a-z0-9])"
                }
                [pscustomobject]@{ Folder = $folder.Name; Pattern = $pattern }
            }
        }
    )

    $files = @(Get-ChildItem -LiteralPath $root -File -ErrorAction Stop)
    if ($null -ne $documents) {
        $foldersToCheck = New-Object 'System.Collections.Generic.Stack[System.IO.DirectoryInfo]'
        $foldersToCheck.Push($documents)
        $files += @(while ($foldersToCheck.Count -gt 0) {
            $current = $foldersToCheck.Pop()
            try {
                $items = @(Get-ChildItem -LiteralPath $current.FullName -ErrorAction Stop)
            } catch {
                Write-Warning "読み取れないフォルダをスキップしました：$($current.Name)"
                continue
            }

            $items | Where-Object { -not $_.PSIsContainer -and -not $_.LinkType }
            foreach ($folder in $items) {
                if ($folder.PSIsContainer -and -not $folder.LinkType -and $folder.Name -notmatch $managedFolder) {
                    $foldersToCheck.Push($folder)
                }
            }
        })
    } else {
        Write-Warning '「ドキュメント」フォルダが見つからないため、OneDrive直下のみ確認します。'
    }

    $number = 0
    $results = @(
        foreach ($file in ($files | Where-Object { -not $_.LinkType } | Sort-Object FullName)) {
            $number++
            $candidates = @()
            $destination = $null
            if ($file.Extension -in @('.url', '.lnk', '.website')) {
                $status = '保留（ショートカット）'
            } else {
                $candidates = @($rules | Where-Object { $file.BaseName -match $_.Pattern } |
                    Select-Object -ExpandProperty Folder -Unique)
                $status = switch ($candidates.Count) {
                    0 { '保留（該当なし）' }
                    1 { '対象' }
                    default { '保留（複数候補）' }
                }
                if ($candidates.Count -eq 1) {
                    $destinationFolder = Join-Path $root $candidates[0]
                    if ($candidates[0] -ne '経費' -and
                        $candidates[0] -notin $rootFolders.Name) {
                        $status = '保留（移動先フォルダなし）'
                    } else {
                        $destination = Join-Path $destinationFolder $file.Name
                        if (Test-Path -LiteralPath $destination) {
                            $status = '保留（移動先に同名あり）'
                        } elseif (-not (Test-Path -LiteralPath $destinationFolder)) {
                            $status = '対象（経費フォルダを作成）'
                        }
                    }
                }
            }
            [pscustomobject]@{
                Number = $number
                Source = $file.FullName
                RelativePath = $file.FullName.Substring($root.Length + 1)
                Target = $destination
                Candidates = $candidates -join ' / '
                Status = $status
            }
        }
    )

    $results | Where-Object { $_.Status -like '対象*' } | Group-Object Target |
        Where-Object Count -gt 1 | ForEach-Object {
            foreach ($item in $_.Group) { $item.Status = '保留（候補間で同名）' }
        }
    if ($results.Count -eq 0) {
        Write-Host '確認対象のファイルはありませんでした。'
        return
    }
    $results | Format-Table @{Label='番号';Expression={$_.Number}},
        @{Label='現在の場所';Expression={$_.RelativePath}},
        @{Label='移動先候補';Expression={$_.Candidates}},
        @{Label='判定';Expression={$_.Status}} -AutoSize -Wrap | Out-Host

    $eligible = @($results | Where-Object { $_.Status -like '対象*' })
    if ($eligible.Count -eq 0) {
        Write-Host '移動できる候補はありません。'
        return
    }
    $selection = Read-Host '移動する番号をカンマ区切りで入力（例：1,3,4）。確認だけならEnter'
    if ([string]::IsNullOrWhiteSpace($selection)) {
        Write-Host 'ファイルは移動していません。'
        return
    }
    $ids = @($selection -split '[,、，]' | ForEach-Object { $_.Trim() } | Select-Object -Unique)
    $chosen = @($eligible | Where-Object { [string]$_.Number -in $ids })
    if ($chosen.Count -ne $ids.Count) {
        Write-Warning '無効な番号または保留の番号が含まれます。移動せず終了します。'
        return
    }
    foreach ($item in $chosen) {
        try {
            $parent = Split-Path -LiteralPath $item.Target
            if ($item.Candidates -eq '経費' -and -not (Test-Path -LiteralPath $parent)) {
                [void][IO.Directory]::CreateDirectory($parent)
            }
            $folder = Get-Item -LiteralPath $parent -ErrorAction Stop
            if (-not $folder.PSIsContainer -or $folder.LinkType) {
                throw '移動先が通常のフォルダではありません。'
            }
            [IO.File]::Move($item.Source, $item.Target)
            Write-Host ("移動済み：{0} → {1}" -f $item.RelativePath, $item.Candidates)
        } catch {
            Write-Warning ("移動できませんでした：{0}（{1}）" -f $item.RelativePath, $_.Exception.Message)
        }
    }
}

OneDriveの整理を1日1回実行

  1. 社用PCのWindows PowerShellに貼り付け、案件ルールと実行時刻を入力します。フォルダ名が不明なら略称だけ入力し、一覧から移動先の番号を選びます。時刻の初期値は12:00です。PCが使える時間を指定してください。
  2. AUTO で1日1回自動移動、REPORT で1日1回候補を記録します。Enterだけで終了すると登録されません。
  3. 「登録済み」と次回実行時刻が表示されたことを確認してください。初回実行後の結果は %LOCALAPPDATA%\OneDriveFileSort\latest.log に記録されます。

OneDrive直下と「ドキュメント」配下を確認します。自動移動は候補が一つのファイルが対象です。複数候補、同名、ショートカットは保留し、Teams・Copilot管理フォルダは除外します。タスクは社用PCにログオンしている間に動きます。勤務先の設定によってタスク登録やスクリプト実行が制限される場合があります。

時刻・案件ルール・モードの変更はこの設定コードを再実行します。一時停止はタスク スケジューラで OneDrive-FileSort-Daily を無効にします。

日次実行のPowerShellコードを表示 
& {
    $ErrorActionPreference = 'Stop'
    $root = $env:OneDriveCommercial
    if ([string]::IsNullOrWhiteSpace($root) -or
        -not (Test-Path -LiteralPath $root -PathType Container)) {
        throw '業務用OneDriveの同期フォルダが見つかりません。'
    }
    $root = (Get-Item -LiteralPath $root).FullName
    $taskName = 'OneDrive-FileSort-Daily'
    $localFolder = Join-Path $env:LOCALAPPDATA 'OneDriveFileSort'
    $scriptPath = Join-Path $localFolder ('daily-sort-' + [Guid]::NewGuid().ToString('N') + '.ps1')
    $logPath = Join-Path $localFolder 'latest.log'
    $powershellPath = Join-Path $env:SystemRoot 'System32\WindowsPowerShell\v1.0\powershell.exe'
    if (-not (Test-Path -LiteralPath $powershellPath -PathType Leaf)) {
        throw 'Windows PowerShellが見つかりません。'
    }

    $resolveAliases = {
        param($entries, $availableFolders)

        $builtIn = @('研修', '経費', 'プロフィール')
        $destinations = @($availableFolders | Where-Object {
            $_.Name -notin (@('ドキュメント', 'Documents') + $builtIn)
        } | Sort-Object Name)
        $confirmed = foreach ($entry in ($entries -split '[,、，]' | Where-Object { $_.Trim() })) {
            $parts = $entry.Trim() -split '[=＝]', 2
            $keyword = $parts[0].Trim()
            $requestedName = $parts[-1].Trim()
            if ($parts.Count -eq 1 -and $keyword -in $builtIn) { continue }
            if (-not $keyword) {
                Write-Warning "キーワードが空のルールはスキップしました：$entry"
                continue
            }

            $folder = $destinations | Where-Object Name -eq $requestedName | Select-Object -First 1
            if ($null -eq $folder) {
                if ($destinations.Count -eq 0) {
                    Write-Warning "選べる案件フォルダがありません：$keyword"
                    continue
                }
                Write-Host ("「{0}」の移動先を、実際のOneDriveフォルダから選んでください。" -f $keyword)
                for ($i = 0; $i -lt $destinations.Count; $i++) {
                    Write-Host ("{0}. {1}" -f ($i + 1), $destinations[$i].Name)
                }
                $choiceText = Read-Host '移動先の番号（Enterはこの案件ルールをスキップ）'
                if ([string]::IsNullOrWhiteSpace($choiceText)) { continue }
                $choice = 0
                if (-not [int]::TryParse($choiceText, [ref]$choice) -or
                    $choice -lt 1 -or $choice -gt $destinations.Count) {
                    Write-Warning "無効な番号のため、案件ルールをスキップしました：$keyword"
                    continue
                }
                $folder = $destinations[$choice - 1]
            }

            if ($folder.Name -match '[,、，]') {
                Write-Warning "カンマを含むフォルダ名は案件ルールに登録できません：$($folder.Name)"
                continue
            }
            Write-Host ("案件ルール：{0} → {1}" -f $keyword, $folder.Name)
            "{0}={1}" -f $keyword, $folder.Name
        }
        $confirmed -join ','
    }
    $rootFolders = @(Get-ChildItem -LiteralPath $root -Directory |
        Where-Object { $_.Name -notmatch 'Teams|Copilot' -and -not $_.LinkType })
    $aliases = Read-Host '案件ルール（例：略称=フォルダ名,フォルダ名）。案件指定なしはEnter'
    $aliases = & $resolveAliases $aliases $rootFolders
    $timeInput = Read-Host '毎日の実行時刻（24時間制 HH:mm）。12:00ならEnter'
    if ([string]::IsNullOrWhiteSpace($timeInput)) { $timeInput = '12:00' }
    if ($timeInput -notmatch '^(?:[01][0-9]|2[0-3]):[0-5][0-9]$') {
        throw '時刻を HH:mm 形式で入力して再実行してください。'
    }
    $clock = $timeInput -split ':'
    $nextRun = [DateTime]::Today.AddHours([int]$clock[0]).AddMinutes([int]$clock[1])
    if ($nextRun -le (Get-Date)) { $nextRun = $nextRun.AddDays(1) }

    $existing = Get-ScheduledTask -TaskPath '\' -TaskName $taskName -ErrorAction SilentlyContinue
    $runnerRegex = [regex]::Escape('"' + (Join-Path $localFolder 'daily-sort-')) + '[0-9a-f]{32}\.ps1"'
    if ($null -ne $existing -and
        ($existing.Actions.Count -ne 1 -or $existing.Actions[0].Arguments -notmatch $runnerRegex)) {
        throw '同じ名前の別のタスクがあります。設定を中止しました。'
    }
    if ($null -ne $existing -and $existing.State -eq 'Running') {
        throw '既存の整理タスクが実行中です。終了後に設定し直してください。'
    }
    Write-Host ("対象：{0} の直下と「ドキュメント」配下" -f $root)
    Write-Host "時刻：毎日 $timeInput（ログオン中）"
    Write-Host ("案件ルール：{0}" -f $aliases)
    if ($null -ne $existing) { Write-Host '既存の日次設定を更新します。' }
    $mode = Read-Host '候補の記録だけなら REPORT、一致したファイルを自動移動するなら AUTO。中止はEnter'
    if ($mode -notin @('REPORT', 'AUTO')) {
        Write-Host '日次設定は変更していません。'
        return

        $answer = Read-Host 'Attendance.ps1 のフルパスを入力してください（Enterで中止）'
        if ([string]::IsNullOrWhiteSpace($answer)) { return }
        $path = $answer.Trim().Trim('"')
    }

    $file = Get-Item -LiteralPath $path
    if ($file.PSIsContainer -or $file.Name -ne 'Attendance.ps1') {
        throw '模擬テスト済みの Attendance.ps1 を指定してください。'
    }

    Write-Host ('対象ファイル: ' + $file.FullName)
    Write-Host '出勤: 会社の正式なログオン連携を使用（NativeStart）。既存の出勤支援タスクがあれば解除します。'
    Write-Host '退勤: 本人がログオンしている平日の17:30に通知。勤務継続時は保留できます。'
    Write-Host '会社が認めた閲覧用の公式入口URLを、この端末内の次の入力欄で設定します。'
    $consent = Read-Host '正式な出勤登録を確認済みで、社内でタスク登録が許可されている場合のみ REGISTER と入力'
    if ($consent -cne 'REGISTER') {
        Write-Host '中止しました。'
        return
    }

    & $file.FullName -Mode Setup -NativeStart -EndTime '17:30'

    $sid = [Security.Principal.WindowsIdentity]::GetCurrent().User.Value
    $prefix = 'LysitheaAttendance-' + $sid
    $tasks = @(Get-ScheduledTask | Where-Object {
        $_.TaskPath -eq '\' -and $_.TaskName -in "$prefix-Start","$prefix-Finish"
    })
    $description = 'LysitheaAttendanceHelper v1 | ' + $file.FullName
    if ($tasks.Count -ne 1 -or $tasks[0].TaskName -ne "$prefix-Finish" -or
        $tasks[0].Description -ne $description -or $tasks[0].State -eq 'Disabled') {
        throw '退勤通知タスクの状態を確認できません。タスク スケジューラで登録内容を確認してください。'
    }
    $tasks | Select-Object TaskName,State
    $info = Get-ScheduledTaskInfo -TaskName "$prefix-Finish" -TaskPath '\'
    Write-Host ('次回の実行時刻: ' + $info.NextRunTime)
    $offDays = Join-Path $env:LOCALAPPDATA 'LysitheaAttendanceHelper\off-days.txt'
    Write-Host ('休日・全日休暇ファイル: ' + $offDays)
    Write-Host '平日の休日・全日休暇を yyyy-MM-dd 形式で1行に1日ずつ保存してください。'
}
